== Stream Logger
- We are getting stream of log messages in this format(log message, timestamp)
```text
log1  8.01
log2  8.05
log3  8.10
log4  8.15
log5  8.20
log6  8.25
log7  8.30
log8  8.35
```
- Print the logs every 10 seconds interval. Make sure not to repeat the logs from prev window

== Solution
- Add all logs into queue

=== 1 Producer 1 Consumer
```cpp
#include <iostream>
#include <thread>
#include <mutex>
#include <queue>
#include <unistd.h>
using namespace std;
mutex m;

using Log = pair<string, float>;
queue<Log> q;

void streamFill(Log log) {
    //lock_guard<mutex> lock(m);
    q.push(log);
}

// Print logs older than the current timestamp and keep newer logs in the queue.
void printer(float timestamp) {
    while (!q.empty()) {
        lock_guard<mutex> lg(m);
        Log l = q.front();
        if (l.second < timestamp) {
            cout << l.first << "\n";
            q.pop();
        }
        else
            break;
    }
}
int main() {
    streamFill(make_pair<string, float>("log1", 8.01));
    streamFill(make_pair<string, float>("log2", 8.05));
    streamFill(make_pair<string, float>("log3", 8.10));
    streamFill(make_pair<string, float>("log4", 8.15));

    float timestamp = 8.10;
    while (1) {
        thread t1(printer, timestamp);
        t1.join();
        timestamp += 10;
        sleep(2);
    }
    return 0;
}
$ g++ Streamlogger_1printer.cpp -lpthread
$ ./a.out 
log1
log2
log3
log4
```

=== n Producer, n Consumer
```cpp
#include <iostream>
#include <thread>
#include <mutex>
#include <queue>
#include <condition_variable>
#include <unistd.h>
using namespace std;
mutex m;
condition_variable cv;
bool producersDone = false;   // This will be done true = when all producers have finished

using Log = pair<string, float>;
queue<Log> q;

void producer(Log log, int tid) {
    unique_lock<mutex> ul(m);
    cv.wait(ul, []
            { return !producersDone; });
    cout << "producer pushed tid:" << tid << "\n";
    q.push(log);
}

void producers_finished() {

    {
        // --------------------------------------------------
        // Important concurrency principle.
        // Keep the critical section as small as possible
        // --------------------------------------------------
        lock_guard<mutex> lock(m);
        producersDone = true;        // All producers have finished
    }

    // Wake all consumers
    cv.notify_all();
}

// Print logs older than the current timestamp and keep newer logs in the queue.
void consumer(float timestamp, int tid) {

    unique_lock<mutex> ul(m);

    // --------------------------------------------------
    // Without this loop consumer will come and go, will not wait for data on q
    // --------------------------------------------------
    while (true) {

        // --------------------------------------------------
        // If the lambda returns true, thread moves forward to execute the code after cv.wait
        // Wait until
        // 1. All producers have finished. (test = true)
        // OR
        // 2. I have finished all logs designated for me
        // --------------------------------------------------
        cv.wait(ul, [&]
        {
            return producersDone ||
                   (!q.empty() && q.front().second <= timestamp);
        });

        while (!q.empty() && q.front().second <= timestamp) {
            Log l = q.front();

            cout << "consumerReading tid:"
                 << tid << ", "
                 << l.first << "\n";

            q.pop();

            // A producer may now be able to add another item.
            // Other consumers should get a chance.
            cv.notify_all();
        }

        // Producers are finished AND this consumer has
        // consumed everything that it is interested in.
        if (producersDone &&
            (q.empty() || q.front().second > timestamp)) {
            break;
        }
    }
}

int main()
{
    // --------------------------------------------------
    // Start consumers FIRST.
    // They will wait because the queue is initially empty.
    // --------------------------------------------------
    vector<thread> tConsumers;
    float timestamp = 8.10;
    tConsumers.emplace_back(consumer, timestamp, 7);
    timestamp += .10;
    tConsumers.emplace_back(consumer, timestamp, 8);
    timestamp += .10;
    tConsumers.emplace_back(consumer, timestamp, 9);

    // --------------------------------------------------
    // Start producers.
    // They now run concurrently with consumers.
    // --------------------------------------------------
    int nthreads = 7;
    thread tproducers[nthreads];
    tproducers[0] = thread(producer, make_pair<string, float>("log1", 8.01), 0);
    tproducers[1] = thread(producer, make_pair<string, float>("log2", 8.05), 1);
    tproducers[2] = thread(producer, make_pair<string, float>("log3", 8.10), 2);
    tproducers[3] = thread(producer, make_pair<string, float>("log4", 8.15), 3);
    tproducers[4] = thread(producer, make_pair<string, float>("log5", 8.20), 4);
    tproducers[5] = thread(producer, make_pair<string, float>("log6", 8.25), 5);
    tproducers[6] = thread(producer, make_pair<string, float>("log6", 8.30), 6);

    // --------------------------------------------------
    // Wait for producers.
    // Consumers are STILL running during this time.
    // --------------------------------------------------
    for (int i = 0; i < nthreads; i++) {
        tproducers[i].join();
    }
    // No producer after this point.

    // Printing. This is not thread safe
    queue<Log> temp = q;
    cout << "printing q\n";
    while (!temp.empty()) {
        Log l = temp.front();
        temp.pop();
        cout << l.first << "," << l.second << "\n";
    }

    // --------------------------------------------------
    // Tell consumers that no more logs will arrive.
    // --------------------------------------------------
    producers_finished();

    // --------------------------------------------------
    // Wait for consumers to finish remaining logs.
    // --------------------------------------------------
    for (int i = 0; i < 3; i++) {
        tConsumers[i].join();
    }

    return 0;
}
```
