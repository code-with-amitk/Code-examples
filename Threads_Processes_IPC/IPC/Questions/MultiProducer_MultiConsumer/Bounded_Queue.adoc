== Bounded Queue
- We have a bounded queue<int> of size=3.
- Producers can write to queue when space is present on queue
- consumers can read when data is present on queue.
- Producers, Consumers execute simultaneously.
- On shutdown, all items from are emptied and then process exits

== Code
=== CPP
==== condition_variable, unique_lock
```cpp
#include <iostream>
#include <queue>
#include <thread>
#include <mutex>
#include <unistd.h>
#include <condition_variable>
using namespace std;

# define CAPACITY 3
bool Done = false;
condition_variable cv;					// pthread_cond_t --> Sequence counter + *ptr_list_of_waiting_threads[i]
int elements = -1;
queue<int> q;

mutex m;								// Underhood --calls-> pthread_mutex_t --calls-> Kernel Space (int futex = 0)

void producer (string pid, int val) {
    unique_lock<mutex> ul(m);			// RAII wrapper object over mutex. unique_lock{ *ptr_to_mutex, bool own }
    cv.wait(ul, []{
       return q.size() < CAPACITY || Done;
    });
    q.push(val);
    cout << "Producer " << pid << " pushed " << val << "\n";
    cv.notify_all();
}
void consumer(string cid) {
    while (true) {
        unique_lock<mutex> ul(m);
        cv.wait(ul, []{
           return !q.empty() || Done;
        });
        if (q.empty() && Done) {
            return;
        }
        cout << "Consumer " << cid << ", val: " << q.front() << "\n";
        q.pop();
        cv.notify_all();
    }
}
void shutdown() {
    {
        unique_lock<mutex> ul(m);
        Done = true;
    }
    cv.notify_all();
}
int main()
{
	/*
	thread > pthread_create() --calls--> clone(){system call} to create struct task_struct
	*/
    thread c1(consumer, "c1");
    thread c2(consumer, "c2");
    thread c3(consumer, "c3");
    sleep(3);
    thread t1(producer, "p1", 1);
    thread t2(producer, "p2", 2);
    thread t3(producer, "p3", 3);
    thread t4(producer, "p4", 4);
    thread t5(producer, "p5", 5);
    
    t1.join();
    t2.join();
    t3.join();
    t4.join();
    t5.join();
    shutdown();
    c1.join();
    c2.join();
    c3.join();
    shutdown();
    cout << "Exiting..";
    
    return 0;
}

$ ./a.out

Producer p4 pushed 4
Consumer c3, val: 4

```

=== Go (goroutine, channel, waitgroups)
```go
package main

import (
	"fmt"
	"sync"
)

var capacity = 3

type ReceiveOnlyChannel = <-chan int
type SendOnlyChannel = chan<- int

// Producer will write on Channel
func producer(pid string, val int, ch SendOnlyChannel, wg *sync.WaitGroup) {
	// Execute wg.Done function after producer
	defer wg.Done()

	// Write value on Channel
	ch <- val
	fmt.Printf("Producer %s pushed %d\n", pid, val)
}

// Consumer will read from Channel
func consumer(cid string, ch ReceiveOnlyChannel, wg *sync.WaitGroup) {
	// Execute wg.Done function after consume
	defer wg.Done()

	// infinite loop to check channel
	for {
		// close the channel, if Producers are not writing data on channel
		// Reading from empty channel will block/deadlock consumer goroutine
		val, ok := <-ch
		if ok {
			fmt.Printf("Consumer %s read %d\n", cid, val)
		} else {
			break
		}
	}
}
func main() {

	ch := make(chan int, capacity)
	var wgConsumer sync.WaitGroup
	var wgProducer sync.WaitGroup
	wgConsumer.Add(3)
	wgProducer.Add(5)
	go consumer("c1", ch, &wgConsumer)
	go consumer("c2", ch, &wgConsumer)
	go consumer("c3", ch, &wgConsumer)

	go producer("p1", 1, ch, &wgProducer)
	go producer("p2", 2, ch, &wgProducer)
	go producer("p3", 3, ch, &wgProducer)
	go producer("p4", 4, ch, &wgProducer)
	go producer("p5", 5, ch, &wgProducer)

	// join()
	wgProducer.Wait()
	close(ch)
	wgConsumer.Wait()

	fmt.Printf("Exiting..")
}
$ go run
```
