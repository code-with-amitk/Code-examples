== ThreadPool
- Create a threadpool of 5 threads. These will keep on running in while(1) loop.
- Threads will monitor a shared queue. When task(ie function pointer) is avaialble on queue, 1 thread will pop the task and execute the function
- All threads will exit based on boolean(stop)

=== CPP
```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <functional>
#include <chrono>

using taskPtr = std::function<void()>;
class ThreadPool
{
public:
    explicit ThreadPool(size_t num_threads = 6) : stop_(false) {
        vecWorkersThreads_.reserve(num_threads);
        for (size_t i = 0; i < num_threads; ++i) {
            std::cout << "Creating thread " << i << "\n";
            vecWorkersThreads_.emplace_back([this, i] {
                while (true) {
                    taskPtr tp;
                    {
                        std::unique_lock<std::mutex> ul(mutex_);
                        
                        // it goes inside if return statement true
                        // That is if process is stopped OR queue is not empty
                        cv_.wait(ul, [this] { 
                            return stop_ || !TasksQueue_.empty(); 
                        });

                        // if process is stopped and Queue is empty, exit
                        if (stop_ && TasksQueue_.empty()) {
                            return;
                        }

                        // Extract task
                        tp = std::move(TasksQueue_.front());
                        TasksQueue_.pop();
                    }

                    // Run task outside lock boundary
                    tp();
                }
            });
        }
    }

    // Submit a void() function or lambda
    void enqueue(taskPtr task) {
        {
            std::unique_lock<std::mutex> ul(mutex_);
            if (stop_) {
                return;
            }
            std::cout << "Pushing task to queue\n";
            TasksQueue_.push(task);
        }
        cv_.notify_one();
    }

    ~ThreadPool() {
        {
            std::unique_lock<std::mutex> ul(mutex_);
            stop_ = true;
        }
        cv_.notify_all();

        for (std::thread &worker : vecWorkersThreads_) {
            if (worker.joinable()) {
                worker.join();
            }
        }
    }

    // Non-copyable
    ThreadPool(const ThreadPool&) = delete;
    ThreadPool& operator=(const ThreadPool&) = delete;

private:
    std::vector<std::thread> vecWorkersThreads_;    //class members are postfixed with _
    std::queue<std::function<void()>> TasksQueue_;
    std::mutex mutex_;
    std::condition_variable cv_;
    bool stop_;
};

// Work function to execute
void process_work(int task_id) {
    std::cout << "[Thread " << std::this_thread::get_id() 
              << "] Executing task " << task_id << "\n";
    std::this_thread::sleep_for(std::chrono::milliseconds(100));
}

int main() {
    // Fixed 6-thread pool
    ThreadPool pool(5);

    // Enqueue 12 tasks using standard std::bind or simple lambdas
    for (int i = 0; i < 10; ++i) {
        pool.enqueue([i]() {
            process_work(i);
        });
    }

    // ThreadPool destructor executes here, flushing the queue before join
    return 0;
}
```

=== Rust
==== OS Thread(thread::spawn()), Communication Queue(std::sync::mpsc::channel)
```rs
use std::sync::{mpsc, Arc, Mutex};
use std::thread;

pub struct ThreadPool {
    workers: Vec<Worker>,
    sender: Option<mpsc::Sender<i32>>,  //Sending end owned by threadpool. receiver moved into threads
}

impl ThreadPool {
    /*
    Create queue=channel on which tasks would be arrive
    Create vector<threads>, assigned to threadpool
    */
    pub fn new(size: usize) -> ThreadPool {
        // Task Queue mpsc(Multi-producer, single-consumer FIFO queue) 
        // Similar to queue in C++
        // Only reciever need to be protected, since Sender is only 1 main thread
        //let (sender, receiver) = std::sync::mpsc::channel::<TaskPtr>();
        let (sender, receiver) = std::sync::mpsc::channel::<i32>();
        let receiver = Arc::new(Mutex::new(receiver));

        // Create workers=threads and add to thread pool
        let mut workers = Vec::with_capacity(size);
        for id in 0..size {
            // Ownership of mpsc::reciever is MOVED to each thread.
            workers.push(Worker::new(id, Arc::clone(&receiver)));
        }
        // Pool of n worker threads, waiting for task to arrive on channel
        ThreadPool {
            workers,
            sender: Some(sender),
        }
    }

    pub fn execute(&self, i: i32)
    {
        if let Some(ref s) = self.sender {
            match s.send(i) {
                Ok(r) => {
                    r
                },
                Err(e) => {
                    println!("{}", e);
                }
            }
        }
    }   
}

// Clean shutdown using Rust's RAII (Drop trait)
impl Drop for ThreadPool {
    fn drop(&mut self) {
        // Drop the sender first; this signals to receivers that no more tasks are coming
        drop(self.sender.take());

        // Join each worker thread
        for worker in &mut self.workers {
            if let Some(thread) = worker.thread.take() {
                thread.join().unwrap();
            }
        }
    }
}

struct Worker {
    id: usize,
    thread: Option<thread::JoinHandle<()>>,
}

impl Worker {
    fn new(id: usize, queue_reciever: Arc<Mutex<mpsc::Receiver<i32>>>) -> Worker {

        Worker {
            id,
            thread: Some(thread::spawn(move || loop {   // Every thread runs in infinite loop
            match queue_reciever.lock().unwrap().recv() {
                Ok(val) => {
                    println!("thread {} Executing Task {}", id, val);
                }
                Err(_) => {
                    // Channel disconnected (sender dropped in ThreadPool::drop)
                    break;
                }
            };
        })),
        }
    }
}

fn main() {
    // Initialize pool with 3 worker threads
    let pool = ThreadPool::new(3);

    for i in 0..5 {
        pool.execute(i);
    }

    // ThreadPool drops automatically at end of main, joining all threads safely
}
```
