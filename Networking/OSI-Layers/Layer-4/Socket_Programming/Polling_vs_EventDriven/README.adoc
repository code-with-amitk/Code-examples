== select() vs poll() vs epoll()
|===
||select|poll|epoll
|What|Polling|polling|event driven
|Limit on number of fds to watch|1024|unlimited|unlimited
|How|O(n). Iterate through all fds(0 to n) to find the active fd|O(n). Same as select|O(1)Only return fds which are ready
|Internally|Bitmasks (fd_set)|Array of pollfd structs|In-kernel Red-Black Tree + Ready List
|===

== select()=Polling. O(n)
=== Code
```cpp
vector<int> client_sockets = { ... }; // N connections

while (true) {
    fd_set read_fds;                    // bitmask
    FD_ZERO(&read_fds);
    
    // 1. Re-populate the bitmask EVERY LOOP for ALL N connections
    int max_fd = 0;
    for (int fd : client_sockets) {
        FD_SET(fd, &read_fds);
        if (fd > max_fd) max_fd = fd;
    }

    // 2. Kernel scans ALL N descriptors from 0 to max_fd
    int activity = select(max_fd + 1, &read_fds, NULL, NULL, NULL);

    // 3. User space must loop through ALL N connections to see who woke up
    for (int fd : client_sockets) {    <<<<<<<<<<<<<<<<<<<<<<<<<< O(n)
        if (FD_ISSET(fd, &read_fds)) {
            handle_client_data(fd);
        }
    }
}
```
=== Limit of 1024 fds
- `fd_set` is fixed-size bit array (bitmap).
- `FD_SETSIZE` was defined as `1024 in <sys/select.h>`, because 1024 file descriptors were considered more than enough for any single process in the early days of UNIX.

== poll()=Polling. O(n)
=== Code
- `poll()` replaces bitmasks with an array of `pollfd structs`. It removes the 1024 file descriptor ceiling, though it still scans the array linearly in the kernel
```c
#include <stdio.h>
#include <unistd.h>
#include <poll.h>

int main() {
    struct pollfd fds[1];
    char buffer[128];

    // 1. Configure the pollfd struct array
    fds[0].fd = STDIN_FILENO;
    fds[0].events = POLLIN;  // Request notification when read data is available

    printf("Waiting for input using poll() (5s timeout)...\n");

    // 2. Pass array and timeout in milliseconds (5000ms = 5s)
    int activity = poll(fds, 1, 5000);

    if (activity < 0) {
        perror("poll error");
    } else if (activity == 0) {
        printf("Timeout! No input received.\n");
    } else {
        // 3. Check revents (returned events) bitfield
        if (fds[0].revents & POLLIN) {
            int n = read(STDIN_FILENO, buffer, sizeof(buffer) - 1);
            buffer[n] = '\0';
            printf("poll() caught input: %s", buffer);
        }
    }
    return 0;
}
```

== epoll()=Event Driven. O(1)
=== Code
```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <sys/epoll.h>

#define PORT 8080
#define MAX_EVENTS 1000  // Maximum events epoll_wait can return at once
#define BUFFER_SIZE 1024

// Helper function to make a socket non-blocking
int make_socket_non_blocking(int fd) {
    int flags = fcntl(fd, F_GETFL, 0);
    if (flags == -1) return -1;
    return fcntl(fd, F_SETFL, flags | O_NONBLOCK);
}

int main() {
    int server_fd, epoll_fd;
    struct sockaddr_in address;
    struct epoll_event ev, events[MAX_EVENTS];

    // 1. Create a TCP listening socket
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd < 0) {
        perror("Socket creation failed");
        return 1;
    }

    // Allow quick re-binding to the port
    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    // Bind to port 8080
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(PORT);

    if (bind(server_fd, (struct sockaddr *)&address, sizeof(address)) < 0) {
        perror("Bind failed");
        close(server_fd);
        return 1;
    }

    // Listen for incoming connections
    if (listen(server_fd, SOMAXCONN) < 0) {
        perror("Listen failed");
        close(server_fd);
        return 1;
    }

    make_socket_non_blocking(server_fd);
    printf("Server listening on port %d...\n", PORT);

    // 2. Create the epoll instance
    epoll_fd = epoll_create1(0);
    if (epoll_fd < 0) {
        perror("epoll_create1 failed");
        close(server_fd);
        return 1;
    }

    // 3. Register the server listening socket to watch for incoming clients (EPOLLIN)
    ev.events = EPOLLIN;
    ev.data.fd = server_fd;
    if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, server_fd, &ev) < 0) {
        perror("epoll_ctl failed for server socket");
        close(server_fd);
        close(epoll_fd);
        return 1;
    }

    // Server loop: Can scale to monitor 1000+ client sockets simultaneously
    while (1) {
        // 4. Wait for events. Kernel returns ONLY active sockets in the 'events' array.
        // It blocks indefinitely (-1 timeout) until at least one socket becomes active.
        int nfds = epoll_wait(epoll_fd, events, MAX_EVENTS, -1);
        if (nfds < 0) {
            perror("epoll_wait error");
            break;
        }

        // 5. Iterate ONLY through the ready/active descriptors returned by the kernel
        for (int i = 0; i < nfds; i++) {
            
            // Case A: The active socket is the server socket (New connection incoming)
            if (events[i].data.fd == server_fd) {
                struct sockaddr_in client_addr;
                socklen_t client_len = sizeof(client_addr);
                
                int client_fd = accept(server_fd, (struct sockaddr *)&client_addr, &client_len);
                if (client_fd < 0) {
                    perror("accept failed");
                    continue;
                }

                make_socket_non_blocking(client_fd);

                // Add this new client socket to our epoll interest list
                ev.events = EPOLLIN | EPOLLET; // Watch for input using Edge Triggered mode
                ev.data.fd = client_fd;
                
                if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, client_fd, &ev) < 0) {
                    perror("epoll_ctl: client_fd");
                    close(client_fd);
                } else {
                    printf("New client connected on socket FD: %d\n", client_fd);
                }
            } 
            
            // Case B: The active socket is an existing client socket (Data sent or disconnected)
            else {
                int client_fd = events[i].data.fd;
                char buffer[BUFFER_SIZE];
                
                // Read from the active network socket
                ssize_t bytes_read = read(client_fd, buffer, sizeof(buffer) - 1);
                
                if (bytes_read > 0) {
                    buffer[bytes_read] = '\0';
                    printf("Received from socket %d: %s", client_fd, buffer);
                    
                    // Optional: Echo back to the client
                    write(client_fd, "Data received!\n", 15);
                } 
                else {
                    // bytes_read <= 0 means client disconnected or an error occurred
                    printf("Client on socket FD %d disconnected.\n", client_fd);
                    
                    // Remove from epoll and close the socket descriptor
                    epoll_ctl(epoll_fd, EPOLL_CTL_DEL, client_fd, NULL);
                    close(client_fd);
                }
            }
        }
    }

    close(server_fd);
    close(epoll_fd);
    return 0;
}
```
=== epoll_wait does not iterate through all N fds
- Unlike `select()` which maintains list of all fds. 
- In `epoll()`, kernel maintains a Ready Linked List (`rdllist`). When network packets arrive, TCP interrupts trigger NIC drivers, which call epoll's internal callback function. That callback instantly appends the specific file descriptor to the kernel's ready list.
- When you call epoll_wait(), the kernel simply checks if that ready list has items.
